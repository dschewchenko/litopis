Integrations / Vue

Vue

@litopis/vue provides the LitopisDatePicker component.

Installation

sh 
npm install @litopis/vue vue
vue 
<script setup lang="ts">
import { LitopisDatePicker } from "@litopis/vue";
import "@litopis/dom/styles/base.css";
</script>

Example

Value: current date

Value

Pass the selected date through value. Listen for value-change to update it.

vue 
<script setup lang="ts">
import { ref } from "vue";
import { LitopisDatePicker, type LitopisDateValue } from "@litopis/vue";

const date = ref<LitopisDateValue | null>(null);
</script>

<template>
  <LitopisDatePicker
    :options="{ calendarMode: 'popover', label: 'Start date' }"
    :value="date"
    @value-change="date = $event"
  />
</template>

Component ref

The component instance exposes controller and getValue().

vue 
<script setup lang="ts">
import { ref } from "vue";
import { LitopisDatePicker } from "@litopis/vue";

const picker = ref<InstanceType<typeof LitopisDatePicker> | null>(null);

function readDate() {
  return picker.value?.getValue() ?? "";
}
</script>

<template><LitopisDatePicker ref="picker" /></template>
